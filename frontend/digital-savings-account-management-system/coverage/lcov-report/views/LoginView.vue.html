<script lang="ts">
import { defineComponent } from "vue";
import { mapActions } from "vuex";
import { toast, Toaster } from "@steveyuowo/vue-hot-toast";
import { FontAwesomeIcon } from '@fortawesome/vue-fontawesome';
import { faArrowRight } from '@fortawesome/free-solid-svg-icons';
import { UserRole } from "@/types/api/user";
import AppLogo from "@/components/AppLogo.vue";
import ForgotPassword from "@/components/ForgotPassword.vue";
 
export default defineComponent({
  name: "LoginView",
  components: {
    Toaster,
    FontAwesomeIcon,
    AppLogo,
    ForgotPassword
  },
  data() {
    return {
      loginEmail: "",
      loginPassword: "",
      showPassword: false,
      isLoading: false,
      forgotModalOpen: false,
      features: [
        "6.5% Interest",
        "Minimum Balance",
        "Lounge Access",
        "Quick Approval"
      ],
      faArrowRight
    };
  },
  methods: {
    ...mapActions("auth", ["login"]),
    async handleLogin() {
      if (!this.loginEmail || !this.loginPassword) {
        toast.error("Please enter both email and password.");
        return;
      }
 
      this.isLoading = true;
 
      try {
        const response = await this.$store.dispatch("auth/login", {
          email: this.loginEmail,
          password: this.loginPassword
        })
 
        const currentUserRole = response.role;
 
        if (currentUserRole === UserRole.ADMIN || currentUserRole === UserRole.SUPER_ADMIN) {
          this.$router.push("/admin");
        } else if (currentUserRole === UserRole.USER) {
          this.$router.push("/me");
        } else {
          toast.error("Unknown user role. Please contact support.");
          this.$router.push("/");
        }
      } catch (error: any) {
        const msg =
          error.response?.data?.message ||
          "Login failed. Please check your credentials.";
        toast.error(msg);
      } finally {
        this.isLoading = false;
      }
    }
  }
});
</script>
