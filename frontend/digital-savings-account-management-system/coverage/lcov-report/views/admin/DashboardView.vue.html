<script lang="ts">
import { defineComponent } from "vue";
import {
  getDashboardStats,
  getApprovedApplications,
} from "@/api/endpoints/admin/admin.api";
import type { AdminApplicationViewResponse, AdminDashboardStatsResponse } from "@/types/adminTypes";
 
export default defineComponent({
  name: "DashboardView",
 
  data() {
    return {
      // State
      stats: {
        managedAssets: 0,
        managedAssetsGrowth: 0,
        approvedUsersCount: 0,
        activeUsersCount: 0,
        newUsersToday: 0,
        averageYield: 0,
        pendingAppsCount: 0,
        adminCount: 0,
        customerCount: 0
      } as AdminDashboardStatsResponse,
 
      approvedApplications: [] as AdminApplicationViewResponse[],
      loading: true,
      error: null as string | null,
      lastUpdated: ""
    };
  },
 
  methods: {
    // Helper functions
    formatCurrency(value: number): string {
      if (!value) return '₹0';
      if (value >= 1000000000) {
        return `₹${(value / 1000000000).toFixed(2)}B`;
      }
      if (value >= 1000000) {
        return `₹${(value / 1000000).toFixed(2)}M`;
      }
      if (value >= 1000) {
        return `₹${(value / 1000).toFixed(1)}K`;
      }
      return `₹${value.toLocaleString()}`;
    },
 
    formatNumber(value: number): string {
      if (!value) return '0';
      return value.toLocaleString();
    },
 
    formatDate(dateString?: string | null): string {
      if (!dateString) return 'N/A';
      try {
        const date = new Date(dateString);
        return date.toLocaleDateString('en-US', {
          year: 'numeric',
          month: 'short',
          day: 'numeric',
          hour: '2-digit',
          minute: '2-digit'
        });
      } catch {
        return dateString;
      }
    },
 
    // Data fetching
    async fetchDashboardData(): Promise<void> {
      this.loading = true;
      this.error = null;
 
      try {
        const [statsData, approvedData] = await Promise.all([
          getDashboardStats(),
          getApprovedApplications()
        ]);
 
        this.stats = statsData;
        this.approvedApplications = approvedData;
 
        this.lastUpdated = new Date().toLocaleTimeString('en-US', {
          hour: '2-digit',
          minute: '2-digit',
          second: '2-digit'
        });
      } catch (err: any) {
 
        if (err.response) {
          if (err.response.status === 404) {
            this.error = 'Dashboard API endpoint not found. Please check backend.';
          } else if (err.response.status === 500) {
            this.error = 'Server error. Please check backend logs.';
          } else {
            this.error = `Error ${err.response.status}: ${err.response.data?.message || 'Failed to load'}`;
          }
        } else if (err.request) {
          this.error = 'Cannot connect to server. Please check if backend is running.';
        } else {
          this.error = 'Failed to load dashboard data. Please try again.';
        }
      } finally {
        this.loading = false;
      }
    },
 
    refreshData(): void {
      this.fetchDashboardData();
    }
  },
 
  mounted(): void {
    this.fetchDashboardData();
  }
});
</script>
